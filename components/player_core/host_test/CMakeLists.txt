# pc_test(<name> <core sources...>): host_test/<name>.c linked with ../src/<sources>
# pc_test_as(<name> <main.c> <core sources...>): the same with another main file (a second build of one test)
function(pc_test name)
  pc_test_as(${name} ${name}.c ${ARGN})
endfunction()
function(pc_test_as name main)
  set(srcs ${main})
  set(core ${ARGN} pc_i18n.c pc_i18n_data.c)   # i18n phase B: TZ() is in every pc_* text
  list(REMOVE_DUPLICATES core)
  foreach(s ${core})
    list(APPEND srcs ${CORE}/src/${s})
  endforeach()
  add_executable(${name} ${srcs})
  target_include_directories(${name} PRIVATE ${CORE}/include ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_options(${name} PRIVATE -Wall -Wextra -Werror)
  target_compile_definitions(${name} PRIVATE FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
  if(PC_ASAN)
    target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
    target_link_options(${name} PRIVATE -fsanitize=address,undefined)
  endif()
  target_link_libraries(${name} m)
  add_test(NAME ${name} COMMAND ${name})
endfunction()
